# Phase 3 Increment 3.2b-1 — Server Master Key + Bootstrap Unlock — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Move the SQLCipher key from a per-login, password-derived value to **one server master key** unwrapped at server start from a bootstrap passphrase, so login authenticates a session without unlocking the database.

**Architecture:** A new pure-INFRA module `app/infra/bootstrap.py` owns a **server keyfile** (a random 256-bit master key wrapped under a bootstrap passphrase, stored outside the encrypted DB). `main.py` unwraps it at startup and sets `app.config["DB_KEY"]`; `get_db()` already opens the DB from that slot. A server-wide **lock gate** returns `503 LOCKED` on `/api/*` (except `/api/health` and `/api/status`) until the key is set. `auth/account.py` and `auth/routes.py` drop all DB-key handling: `create_account` writes a minimal `auth.json` (still single-account here), `authenticate` is a bcrypt check only, login/logout no longer set or clear `DB_KEY` (logout must not lock the whole server). Behavior *changes* — this is the security spine of Phase 3.

**Tech Stack:** Python 3.11, Flask, SQLCipher (`sqlcipher3`), `cryptography` (scrypt KEK + AES-GCM wrap, already used by `auth/account.py`), bcrypt, pytest.

## Bootstrap passphrase / keyfile contract (reference for all tasks)

- **Env:** `LP_BOOTSTRAP_PASSPHRASE` supplies the passphrase; `main.py` falls back to a console `getpass` prompt. No PyWebView dialog this increment.
- **Keyfile:** `<APP_DATA_DIR>/server.key`, JSON `{ "kdf_salt": b64, "nonce": b64, "wrapped_key": b64 }`, perms `0600` (best-effort on Windows). Holds the master key wrapped under `derive_kek(passphrase, salt)` via AES-GCM. Gitignored, never committed.
- **First run** (`server.key` absent): generate a random 32-byte master key, wrap it, write the keyfile, return the key. **Later runs:** unwrap and return; a wrong passphrase / corrupt keyfile raises `BootstrapError`.
- Bootstrap does **not** open or create the DB. The first `get_db()` connects with the master key and the migration seam (ADR 0002) builds the baseline on a fresh DB.

```python
"""Server master-key bootstrap (INFRA).

The server holds ONE master SQLCipher key for the whole process, unwrapped at
startup from a bootstrap passphrase (ADR 0003 part 2). The wrapped key lives in a
keyfile OUTSIDE the encrypted DB; the raw key exists only in process memory.

Pure INFRA: manages the keyfile and returns the raw key. It does NOT open the
database -- creation stays lazy in api.guards.get_db via the migration seam
(ADR 0002) -- and imports no Flask.
"""

from __future__ import annotations

import base64
import binascii
import json
import logging
import os
import stat
import tempfile

from cryptography.exceptions import InvalidTag

from app.infra import config, crypto

logger = logging.getLogger(__name__)

KEYFILE_NAME = "server.key"


class BootstrapError(Exception):
    """The master key could not be unwrapped: missing/corrupt keyfile or wrong passphrase."""


def keyfile_path():
    return config.app_data_dir() / KEYFILE_NAME


def is_initialized() -> bool:
    return keyfile_path().exists()


def _b64e(b: bytes) -> str:
    return base64.b64encode(b).decode("ascii")


def _b64d(s: str) -> bytes:
    return base64.b64decode(s.encode("ascii"), validate=True)


def _write_keyfile(record: dict) -> None:
    path = keyfile_path()
    path.parent.mkdir(parents=True, exist_ok=True)
    fd, tmp = tempfile.mkstemp(dir=path.parent, suffix=".tmp")
    try:
        with os.fdopen(fd, "w", encoding="utf-8") as fh:
            json.dump(record, fh)
            fh.flush()
            os.fsync(fh.fileno())
        os.replace(tmp, path)
    except Exception:
        try:
            os.unlink(tmp)
        except OSError:
            logger.warning("could not remove temp keyfile %s after a failed write", tmp)
        raise
    try:
        os.chmod(path, stat.S_IRUSR | stat.S_IWUSR)  # 0600; best-effort (no-op semantics on Windows)
    except OSError:
        logger.warning("could not restrict permissions on %s", path)


def _initialize(passphrase: str) -> bytes:
    master_key = os.urandom(config.DB_KEY_BYTES)
    salt = os.urandom(config.SCRYPT_SALT_BYTES)
    kek = crypto.derive_kek(passphrase, salt)
    nonce, wrapped = crypto.wrap_key(kek, master_key)
    _write_keyfile({"kdf_salt": _b64e(salt), "nonce": _b64e(nonce), "wrapped_key": _b64e(wrapped)})
    return master_key


def _unlock_existing(passphrase: str) -> bytes:
    try:
        with open(keyfile_path(), "r", encoding="utf-8") as fh:
            rec = json.load(fh)
        salt, nonce, wrapped = _b64d(rec["kdf_salt"]), _b64d(rec["nonce"]), _b64d(rec["wrapped_key"])
    except (OSError, json.JSONDecodeError, KeyError, TypeError, ValueError, binascii.Error) as exc:
        raise BootstrapError("corrupt or unreadable keyfile") from exc
    try:
        kek = crypto.derive_kek(passphrase, salt)
        master_key = crypto.unwrap_key(kek, nonce, wrapped)
    except (InvalidTag, ValueError) as exc:
        raise BootstrapError("wrong bootstrap passphrase") from exc
    if len(master_key) != config.DB_KEY_BYTES:
        raise BootstrapError("keyfile holds a malformed key")
    return master_key


def unlock(passphrase: str) -> bytes:
    """Return the raw master key: generate + persist it on first run, else unwrap it.

    Raises BootstrapError on an empty passphrase, a wrong passphrase, or a corrupt keyfile.
    """
    if not passphrase:
        raise BootstrapError("no bootstrap passphrase provided")
    if is_initialized():
        return _unlock_existing(passphrase)
    return _initialize(passphrase)
```

```python
import pathlib

SRC = pathlib.Path(__file__).resolve().parents[2] / "app" / "infra" / "bootstrap.py"
FORBIDDEN_PREFIXES = ("flask", "app.auth", "app.api", "app.datasets", "app.targets",

def _imported_names(tree):
    for node in ast.walk(tree):
        if isinstance(node, ast.Import):
            for a in node.names:
                yield a.name
        elif isinstance(node, ast.ImportFrom) and node.module:
            yield node.module
```

```python
def install_lock_gate(app):
    """Return 503 LOCKED for every /api/* route until the server master key is set.

    /api/health and /api/status stay reachable so a client can observe the locked
    state (ADR 0003 part 2). This is the server-wide unlock gate -- distinct from
    login_required, which gates a per-session login once the server is unlocked.
    Non-/api paths (the SPA shell/assets) are never gated, so a locked LAN server
    can still serve the page that reports it is locked.
    """
    _exempt = {"/api/health", "/api/status"}

    @app.before_request
    def _require_unlocked():
        path = request.path
        if not path.startswith("/api/") or path in _exempt:
            return None
        if current_app.config.get("DB_KEY") is None:
            return err("LOCKED", "server is locked", 503)
        return None
```

(`request`, `current_app`, and `err` are already imported in `guards.py`.)

```python
def create_account(username: str, password: str) -> None:
    """Create the single local account. The DB key is the server master key
    (bootstrap, ADR 0003 pt2), so setup only records the credential; the DB itself
    is built lazily by the first get_db() via the migration seam."""
    if account_exists():
        raise AccountExistsError("an account already exists")
    _check_password_length(password)
    _save({
        "username": username,
        "bcrypt_hash": crypto.hash_password(password),
        "created_at": _now(),
    })


def authenticate(username: str, password: str) -> None:
    """Verify credentials against auth.json. Raises AuthError on any mismatch.
    No DB key is derived -- the server already holds the master key."""
    if not account_exists():
        raise AuthError("no account")
    try:
        rec = _load()
        stored_username = rec["username"]
        bcrypt_hash = rec["bcrypt_hash"]
    except (OSError, json.JSONDecodeError, KeyError, TypeError, ValueError):
        raise AuthError("bad credentials") from None
    try:
        username_ok = hmac.compare_digest(
            username.encode("utf-8"), stored_username.encode("utf-8"))
        password_ok = crypto.verify_password(password, bcrypt_hash)
    except (AttributeError, TypeError, ValueError):
        raise AuthError("bad credentials") from None
    if not (username_ok and password_ok):
        raise AuthError("bad credentials")


def change_password(old_password: str, new_password: str) -> None:
    if not account_exists():
        raise AuthError("no account")
    _check_password_length(new_password)
    try:
        rec = _load()
        username = rec["username"]
    except (OSError, json.JSONDecodeError, KeyError, TypeError, ValueError):
        raise AuthError("bad credentials") from None
    authenticate(username, old_password)  # raises AuthError if wrong
    rec["bcrypt_hash"] = crypto.hash_password(new_password)
    _save(rec)
```

```python
def resolve_and_unlock():
    """Unwrap the server master key at startup. Passphrase from LP_BOOTSTRAP_PASSPHRASE,
    else a console prompt. Returns the key, or None (server starts locked) on failure."""
    passphrase = os.environ.get("LP_BOOTSTRAP_PASSPHRASE")
    if not passphrase:
        try:
            passphrase = getpass.getpass("Bootstrap passphrase: ")
        except (EOFError, KeyboardInterrupt):
            passphrase = ""
    try:
        return bootstrap.unlock(passphrase)
    except bootstrap.BootstrapError as exc:
        logging.getLogger(__name__).error("server locked: %s", exc)
        return None
```
